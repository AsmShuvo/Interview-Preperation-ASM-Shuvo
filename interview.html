Problem 1: Bipartite Graph Coloring

Description: Given an undirected graph, determine if it is bipartite. If it is bipartite, print one of the two sets of vertices that form the bipartition. If the graph is not bipartite, print "-1".

Code (C++)


                  #include <bits/stdc++.h>
using namespace std;

bool ok = true;

// Time: O(v+e) Space: O(v+e)
void dfs(vector<vector<int>> &g, vector<bool> &vis, vector<int> &color, int v, int p)
{
    if (!ok)
        return;
    vis[v] = true;
    if (p >= 0)
        color[v] = !color[p];
    for (auto u : g[v])
    {
        if (vis[u] && u != p && color[u] == color[v])
        {
            ok = false;
            return;
        }
        if (!vis[u])
        {
            dfs(g, vis, color, u, v);
        }
    }
}

int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        ok = true;
        int v, e;
        cin >> v >> e;
        vector<vector<int>> g(v + 1);
        vector<bool> vis(v + 1, false);
        vector<int> color(v + 1);
        for (int i = 0; i < e; i++)
        {
            int u, v;
            cin >> u >> v;
            g[u].push_back(v);
            g[v].push_back(u);
        }
        for (int i = 1; i <= v; i++)
        {
            if (!vis[i])
                dfs(g, vis, color, i, -1);
        }
        if (!ok)
        {
            cout << -1 << endl;
        }
        else
        {
            for (int i = 1; i <= v; i++)
            {
                if (color[i])
                    cout << i << " ";
            }
            cout << endl;
        }
    }
    return 0;
}

Problem 3: Catch the Stationary Piece (Knight's Move)

Description: There is a mobile piece and a stationary piece on an N x M chessboard. You need to capture the stationary piece by moving the mobile piece with the minimum number of moves. The mobile piece moves like a Knight.

Code (cpp)


#include <bits/stdc++.h>
using namespace std;

int n, m;
int sr, sc, dr, dc;
vector<int> dx = {2, 2, -2, -2, 1, -1, 1, -1};
vector<int> dy = {-1, 1, -1, 1, -2, -2, 2, 2};

int bfs()
{
    vector<vector<int>> vis(n + 1, vector<int>(m + 1));
    vis[sr][sc] = 1;
    queue<vector<int>> q;
    // r c step
    q.push({sr, sc, 0});
    while (!q.empty())
    {
        auto cur = q.front();
        q.pop();
        int x = cur[0], y = cur[1], step = cur[2];
        if (x == dr && y == dc)
            return step; // found
        for (int i = 0; i < 8; i++)
        {
            int nx = x + dx[i];
            int ny = y + dy[i];
            if (nx < 1 || nx > n || ny < 1 || ny > m || vis[nx][ny])
                continue;
            vis[nx][ny] = 1;
            q.push({nx, ny, step + 1});
        }
    }
    return -1;
}

int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        cin >> n >> m;
        cin >> sr >> sc >> dr >> dc;
        cout << bfs() << endl;
    }
}

Problem 5: Crow and Overflowing Pots

Description: There are N pots. Every pot has some water in it and is associated with an overflow number O_i. The crow wants to overflow at least K pots. Find the minimum number of stones the crow needs to use to guarantee that K pots overflow in the worst-case scenario.

Code (C++)

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int tl;
    if (!(cin >> tl)) return 0;
    while (tl--)
    {
        int n, m, total_time;
        cin >> n >> m >> total_time;
        int steps = total_time / 10;
        
        vector<vector<pair<int, double>>> g(n + 1);
        for (int i = 0; i < m; i++)
        {
            int u, v;
            double p;
            cin >> u >> v >> p;
            g[u].push_back({v, p});
        }
        
        vector<double> prob(n + 1, 0.0);
        prob[1] = 1.0;

        for (int s = 0; s < steps; s++)
        {
            vector<double> nextProb(n + 1, 0.0);
            for (int u = 1; u <= n; u++)
            {
                if (prob[u] > 0)
                {
                    for (auto &e : g[u])
                    {
                        int v = e.first;
                        double pb = e.second; 
                        nextProb[v] += prob[u] * pb;
                    }
                }
            }
            prob = nextProb;
        }

        int lastNode = -1;
        double mxProb = -1.0; 

        for (int i = 1; i <= n; i++)
        {
            if (prob[i] > mxProb)
            {
                mxProb = prob[i];
                lastNode = i;
            }
        }

        if (lastNode == -1 || mxProb <= 0.0)
            cout << "impossible" << endl;
        else
            cout << lastNode << " " << fixed << setprecision(6) << mxProb << endl;
    }
    return 0;
}

Problem 6: Doctor Probability

Description: A Doctor travels between divisions (nodes) in a directed graph where edge weights are probabilities. He stays 10 mins at each division. Given time T, find the division with the highest probability where he will be staying.

Code (C++)

#include <iostream>
#include <vector>
#include <iomanip>

using namespace std;

void solve() {
    int N, E, T;
    cin >> N >> E >> T;
    
    // Adjacency list: u -> {v, prob}
    vector<vector<pair<int, double>>> adj(N + 1);
    for (int i = 0; i < E; ++i) {
        int u, v;
        double p;
        cin >> u >> v >> p;
        adj[u].push_back({v, p});
    }

    // Probability array
    vector<double> prob(N + 1, 0.0);
    prob[1] = 1.0; // Assume start at node 1

    int steps = T / 10;
    
    for (int s = 0; s < steps; ++s) {
        vector<double> next_prob(N + 1, 0.0);
        bool any_move = false;
        for (int u = 1; u <= N; ++u) {
            if (prob[u] > 0) {
                if (!adj[u].empty()) {
                    for (auto& edge : adj[u]) {
                        int v = edge.first;
                        double p = edge.second;
                        next_prob[v] += prob[u] * p;
                    }
                    any_move = true; 
                } 
            }
        }
        prob = next_prob;
    }

    int result_node = 0;
    double max_p = -1.0;
    
    for (int i = 1; i <= N; ++i) {
        if (prob[i] > max_p) {
            max_p = prob[i];
            result_node = i;
        }
    }
    
    cout << result_node << " " << fixed << setprecision(6) << max_p << endl;
}

int main() {
    int t;
    cin >> t;
    while(t--) {
        solve();
    }
    return 0;
}

Problem 1: Remove Duplicates from a Linked List

Description: Remove all duplicate elements from a singly linked list such that each value appears only once.

Code (C++)

/*
✅ Problem Statement: Remove Duplicates from a Linked List

You are given the head of a singly linked list.
The linked list may contain duplicate values.

Your task is to remove all duplicate elements such that each value appears only once in the linked list.

After removing duplicates, return the head of the modified linked list.
*/
Node* RemoveDuplicates(Node* head){
    unordered_set<int> seen;

    Node* dummy = new Node(0);
    dummy->next = head;

    Node* prev = dummy;
    Node* cur  = head;

    while(cur){
        if(seen.count(cur->val)){
            prev->next = cur->next;
            Node* tmp = cur;
            cur = cur->next;
            delete tmp;
        } else {
            seen.insert(cur->val);
            prev = cur;
            cur = cur->next;
        }
    }

    Node* ans = dummy->next;
    delete dummy;
    return ans;
}

General Technical Questions (Q&A)

1. What is Concurrency Control in Databases?

Concurrency control manages simultaneous operations on a database without them conflicting with each other. It ensures that database transactions are performed concurrently without violating data integrity. Key concepts include ACID properties, Locking (Shared/Exclusive), and Isolation Levels (Read Committed, Serializable, etc.).

2. What is function overriding in OOP? How is it different from overloading?

Overriding: Occurs when a derived class provides a specific implementation of a method that is already defined in its base class. It is a runtime polymorphism feature (same signature).

Overloading: Occurs when multiple methods in the same class have the same name but different parameters (different signature). It is a compile-time feature.

3. What is inheritance in OOP?

Inheritance is a mechanism where a new class (child/derived) acquires the properties and behaviors (fields and methods) of an existing class (parent/base). It promotes code reusability and establishes an "Is-A" relationship.

4. What is database normalization? Benefits and common forms?

Normalization is the process of organizing data in a database to reduce redundancy and improve data integrity.

  • 1NF: Atomic values (no repeating groups).
  • 2NF: 1NF + No partial dependency.
  • 3NF: 2NF + No transitive dependency.
  • BCNF: A stricter version of 3NF.

Benefits: Minimal data duplication, easier maintenance, and smaller database size.

5. How can you generate random numbers in C++?

Using the <random> library is the modern way: std::mt19937 generator and distributions like std::uniform_int_distribution. The legacy way is using `rand()` and `srand(time(0))`, but it has lower quality randomness.

6. Basic Linux commands and use cases?
  • ls: List directory contents.
  • cd: Change directory.
  • grep: Search for patterns in files.
  • chmod: Change file permissions.
  • ps: Display currently running processes.
  • top: Display real-time system resource usage.
7. What is a deadlock? How to prevent it?

A deadlock is a situation where two or more processes are unable to proceed because each is waiting for the other to release a resource.
Prevention: Ensure at least one of the 4 Coffman conditions cannot hold (Mutual Exclusion, Hold and Wait, No Preemption, Circular Wait). For example, imposing a strict ordering on resource acquisition prevents Circular Wait.

8. How to compare two text files to see if they are identical?

Command Line: Use `diff file1.txt file2.txt` or `cmp file1.txt file2.txt` in Linux/Unix. In Windows, use `FC file1.txt file2.txt`.
Programmatically: Read both files byte-by-byte or line-by-line. If any byte/line differs, they are not identical. Alternatively, compute and compare their hash checksums (MD5/SHA256).

9. What happens when you type google.com and press enter?
  1. DNS Query: Browser checks cache, then asks DNS server for IP of google.com.
  2. TCP Handshake: Browser establishes a TCP connection with the server (SYN, SYN-ACK, ACK).
  3. HTTP Request: Browser sends a GET request to the server.
  4. Server Processing: Server processes request and sends back an HTTP response (HTML).
  5. Rendering: Browser parses HTML, CSS, JS and renders the page.
10. What happens when you copy a file? Are they the same?

The operating system reads the data bits from the source location on the storage medium and writes an identical sequence of bits to the destination location. It also creates a new directory entry for the copy.

Are they the same? The content is bit-for-bit identical. However, file metadata (creation time, file path, owner, permissions) will usually differ.
